34	// Map Supabase error messages → stable i18n keys (Epic A error-code contract)
35	function mapAuthError(message: string): string {
36	  const m = message.toLowerCase()
37	  // A banned account gets the wrong-credentials text (Task 923 R40, D21): Supabase Auth answers `user_banned` before it checks the
38	  // password, so a distinct text would tell a visitor that an account exists and is blocked (OWASP ASVS 5.0 6.3.8).
39	  if (m.includes('invalid login credentials') || m.includes('invalid credentials') || m.includes('user is banned')) return 'error_invalid_credentials'
40	  if (m.includes('user already registered') || m.includes('already registered') || m.includes('already exists')) return 'error_email_exists'
41	  if (m.includes('password should be at least') || m.includes('weak password')) return 'error_weak_password'
42	  if (m.includes('email not confirmed') || m.includes('not confirmed')) return 'error_email_not_confirmed'
43	  if (m.includes('rate limit') || m.includes('too many')) return 'error_rate_limit'
44	  return 'error_generic'
45	}
