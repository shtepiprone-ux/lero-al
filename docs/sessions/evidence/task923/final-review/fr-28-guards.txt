## guard_users_protected_columns
CREATE OR REPLACE FUNCTION public.guard_users_protected_columns()
 RETURNS trigger
 LANGUAGE plpgsql
 SET search_path TO 'public', 'pg_temp'
AS $function$
begin
  -- service_role, the table owner and SECURITY DEFINER functions are trusted; an active administrator acting through a session is too
  if current_user not in ('authenticated', 'anon') or public.caller_is_admin() then return new; end if;
  if tg_op = 'INSERT' then
    -- a session may only ever insert a plain, active, unverified, never-sanctioned user (no policy allows an insert today; this does not depend on that)
    if new.role::text <> 'user'
    or coalesce(new.is_verified, false)
    or coalesce(new.status::text, 'active') <> 'active'
    or new.block_reason is not null
    or new.suspended_until is not null
    or new.deleted_at is not null then
      raise exception 'protected column' using errcode = '42501';
    end if;
    return new;
  end if;
  if new.id is distinct from old.id
  or new.role is distinct from old.role
  or new.is_verified is distinct from old.is_verified
  or new.status is distinct from old.status
  or new.block_reason is distinct from old.block_reason
  or new.suspended_until is distinct from old.suspended_until
  or new.deleted_at is distinct from old.deleted_at
  or new.public_id is distinct from old.public_id then
    raise exception 'protected column' using errcode = '42501';
  end if;
  return new;
end $function$

## guard_listings_protected_columns
CREATE OR REPLACE FUNCTION public.guard_listings_protected_columns()
 RETURNS trigger
 LANGUAGE plpgsql
 SET search_path TO 'public', 'pg_temp'
AS $function$
begin
  if current_user not in ('authenticated', 'anon') or public.caller_is_admin() then return new; end if;
  if tg_op = 'INSERT' then
    if new.status is distinct from 'pending'::listing_status
    or coalesce(new.is_premium, false)
    or new.premium_until is not null
    or coalesce(new.views_count, 0) <> 0
    or (new.expires_at is not null and new.expires_at > now() + interval '61 days')
    or new.created_at is distinct from now()
    or new.slug is null or new.slug !~ '^[a-z0-9]+(-[a-z0-9]+)*$' or length(new.slug) > 120 then
      raise exception 'protected column' using errcode = '42501';
    end if;
    new.public_id := nextval('public.listings_public_id_seq'::regclass);   -- whatever the client sent: the next value of the sequence
    return new;
  end if;
  if new.is_premium is distinct from old.is_premium
  or new.premium_until is distinct from old.premium_until
  or new.views_count is distinct from old.views_count
  or new.expires_at is distinct from old.expires_at
  or new.public_id is distinct from old.public_id
  or new.user_id is distinct from old.user_id
  or new.status is distinct from old.status
  or new.created_at is distinct from old.created_at
  or new.slug is distinct from old.slug then
    raise exception 'protected column' using errcode = '42501';
  end if;
  return new;
end $function$
