+// Scans forward from `openIndex` (which must point at a `{`) and returns the index of its
+// matching `}`, skipping brace characters that appear inside string/template literals. Returns -1
+// if the content ends before the brace balances (malformed/truncated input).
+function findMatchingBraceIndex(content, openIndex) {
+  let depth = 0;
+  let inString = null;
+  for (let i = openIndex; i < content.length; i++) {
+    const ch = content[i];
+    if (inString) {
+      if (ch === '\\') { i++; continue; }
+      if (ch === inString) inString = null;
+      continue;
+    }
+    if (ch === "'" || ch === '"' || ch === '`') { inString = ch; continue; }
+    if (ch === '{') depth++;
+    else if (ch === '}') {
+      depth--;
+      if (depth === 0) return i;
+    }
+  }
+  return -1;
+}
