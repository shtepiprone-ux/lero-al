352	function ModalDeleteConfirmPasswordSection({ locale }: { locale: string }) {
353	  const [opened, setOpened] = useState(false)
354	  const [loading, setLoading] = useState(false)
355	  const [password, setPassword] = useState('')
356	  const [error, setError] = useState<string | undefined>(undefined)
357	  const t = (key: string) => storyT(locale, `storybook.mantine.${key}`)
358	
359	  return (
360	    <MantineStorySection label={t('modal_caption_delete')}>
361	      <Button
362	        variant="default"
363	        onClick={() => {
364	          setPassword('')
365	          setError(undefined)
366	          setOpened(true)
367	        }}
368	      >
369	        {t('action_delete')}
370	      </Button>
371	      <MantineDeleteConfirmModal
372	        opened={opened}
373	        onClose={() => setOpened(false)}
374	        title={t('dialog_confirm_title')}
375	        message={t('dialog_confirm_body')}
376	        confirmLabel={t('action_delete')}
377	        cancelLabel={t('action_cancel')}
378	        loading={loading}
379	        passwordConfirm={{
380	          label: storyT(locale, 'cabinet.delete_account_password_confirm'),
381	          value: password,
382	          onChange: (value) => { setPassword(value); setError(undefined) },
383	          error,
384	          showLabel: storyT(locale, 'common.show_password'),
385	          hideLabel: storyT(locale, 'common.hide_password'),
386	        }}
387	        onConfirm={() => {
388	          setLoading(true)
389	          window.setTimeout(() => {
390	            setLoading(false)
391	            setError(storyT(locale, 'cabinet.password_error_invalid_current'))
392	          }, 800)
393	        }}
394	      />
395	    </MantineStorySection>
396	  )
397	}
