describe('setModeratorPermission and the history', () => {
  const seed = () => createFakeDb({ role_permissions: [{ role: 'moderator', permission_key: 'x', allowed: false }], role_permission_events: [], users: [] })
  it('an unexpected failure answers save_failed, never a success', async () => {
    h.throwOnClient = true
    const { setModeratorPermission } = await import('@/modules/admin/actions/permissions')
    const { PERMISSION_KEYS } = await import('@/lib/auth/permissionKeys')
    expect(await setModeratorPermission(PERMISSION_KEYS[0], true)).toEqual({ error: 'save_failed' })
  })
  it('a failed audit insert is logged and the saved change still answers success; a good one logs nothing', async () => {
    const { setModeratorPermission } = await import('@/modules/admin/actions/permissions')
    const { PERMISSION_KEYS } = await import('@/lib/auth/permissionKeys')
    const db = seed()
    h.client = db.client
    expect(await setModeratorPermission(PERMISSION_KEYS[0], true)).toEqual({})
    expect(errorSpy).not.toHaveBeenCalled()
    const failing = seed()
    failing.fail({ table: 'role_permission_events', op: 'insert', error: { message: 'down' } })
    h.client = failing.client
    expect(await setModeratorPermission(PERMISSION_KEYS[0], true)).toEqual({})
    expect(errorSpy).toHaveBeenCalledWith('setModeratorPermission event insert failed', expect.objectContaining({ key: PERMISSION_KEYS[0] }))
  })
})

describe('applyListingTransition effects', () => {
  const actor = { userId: 'admin-1', role: 'admin' } as never
  const seed = (extra: Record<string, unknown> = {}) => createFakeDb({ listings: [{ id: 'l1', status: 'active', slug: 's1', title: 'Flat', user_id: 'owner-1', ...extra }] })
  it('refreshes the index and the detail page in every locale, and notifies the owner with the title and link', async () => {
    const { applyListingTransitionByStatus } = await import('@/modules/listings/actions/applyListingTransition')
    const { routing } = await import('@/i18n/routing')
    const db = seed()
    expect(await applyListingTransitionByStatus('l1', 'sold', actor, db.client as never)).toEqual({ ok: true, nextStatus: 'sold', listingId: 'l1' })
    for (const locale of routing.locales) {
      expect(h.revalidatePath).toHaveBeenCalledWith(`/${locale}/listings`, 'page')
      expect(h.revalidatePath).toHaveBeenCalledWith(`/${locale}/listings/s1`, 'page')
    }
    expect(h.createNotification).toHaveBeenCalledWith(expect.objectContaining({ userId: 'owner-1', title: 'Flat', link: '/listings/s1' }))
  })
  it('no slug: only the index is refreshed; no title: the notice carries the id; no link', async () => {
    const { applyListingTransitionByStatus } = await import('@/modules/listings/actions/applyListingTransition')
    const { routing } = await import('@/i18n/routing')
    await applyListingTransitionByStatus('l1', 'sold', actor, seed({ slug: null, title: null }).client as never)
    expect(h.revalidatePath).toHaveBeenCalledTimes(routing.locales.length)
    expect(h.createNotification).toHaveBeenCalledWith(expect.objectContaining({ title: 'l1', link: undefined }))
  })
  it('the actor who owns the listing is not notified', async () => {
    const { applyListingTransitionByStatus } = await import('@/modules/listings/actions/applyListingTransition')
    await applyListingTransitionByStatus('l1', 'sold', actor, seed({ user_id: 'admin-1' }).client as never)
    expect(h.createNotification).not.toHaveBeenCalled()
  })
  it('a lost race: conflict, nothing refreshed or notified, and the log names the actor', async () => {
    const { applyListingTransitionByStatus } = await import('@/modules/listings/actions/applyListingTransition')
    const db = seed()
    db.fail({ table: 'listings', op: 'update', error: null })
    expect((await applyListingTransitionByStatus('l1', 'sold', actor, db.client as never)).ok).toBe(false)
    expect(h.revalidatePath).not.toHaveBeenCalled()
    expect(warnSpy).toHaveBeenCalledWith(expect.any(String), expect.objectContaining({ actor: 'admin-1', rows: 0 }))
  })
})

describe('AdminListingsPage past its guard', () => {
  const rows = [
    { id: 'a', title: 'Sea villa', status: 'active', is_premium: true, created_at: '2026-01-03', expires_at: '2999-01-01', user_id: 'u1' },
    { id: 'b', title: 'City flat', status: 'pending', is_premium: false, created_at: '2026-01-02', expires_at: null, user_id: 'u2' },
    { id: 'c', title: 'Old villa', status: 'active', is_premium: false, created_at: '2026-01-01', expires_at: '2000-01-01', user_id: 'u3' },
  ]
  async function open(sp: Record<string, string> = {}, rpc: unknown = { data: [] }) {
    const db = createFakeDb({ listings: rows, users: [] })
    h.client = { ...db.client, rpc: vi.fn(async () => rpc) }
    const { default: Page } = await import('@/app/admin/listings/page')
    const { render } = await import('@testing-library/react')
    render(await Page({ searchParams: Promise.resolve(sp) }) as never)
    return { db, props: h.tableProps as Record<string, unknown>, main: db.log.find(e => e.columns?.includes('owner:users')) }
  }
  it('default: every row newest first, 25 a page from zero, the locale resolved, audit counts head only', async () => {
    const { db, props, main } = await open()
    expect(h.getAdminLocale).toHaveBeenCalledTimes(1)
    expect((props.listings as { id: string }[]).map(r => r.id)).toEqual(['a', 'b', 'c'])
    expect(main?.orders).toEqual([{ column: 'created_at', ascending: false }])
    expect(main?.range).toEqual([0, 24])
    expect(props).toMatchObject({ total: 3, page: 1, perPage: 25, activeStatus: '', searchQuery: '', activeTab: 'all', activeVisibility: '', activeReason: '' })
    const audits = db.log.filter(e => e.head === true)
    expect(audits).toHaveLength(3)
    // 2, not 1: this prototype's fake does not evaluate or() and counts the active row that has not expired. The fake of R6 does, and the real test expects 1.
    expect(props.auditCounts).toEqual({ total: 2, expired: 1, noExpiry: 0 })
  })
  it('the premium tab and the status filter narrow the rows; another tab does not', async () => {
    expect(((await open({ tab: 'premium' })).props.listings as { id: string }[]).map(r => r.id)).toEqual(['a'])
    expect(((await open({ tab: 'other' })).props.listings as unknown[])).toHaveLength(3)
    expect(((await open({ status: 'pending' })).props.listings as { id: string }[]).map(r => r.id)).toEqual(['b'])
  })
  it('the page parameter moves the range, and a bad page falls back to the first', async () => {
    expect((await open({ page: '3' })).main?.range).toEqual([50, 74])
    expect((await open({ page: '-4' })).main?.range).toEqual([0, 24])
  })
  it('the visibility filter reaches the query', async () => {
    expect(((await open({ visibility: 'visible' })).props.listings as { id: string }[]).map(r => r.id)).toEqual(['a'])
  })
  it('the search filter: words in the title, owner matches by name and e-mail, an exact id', async () => {
    const { main } = await open({ q: 'sea villa' }, { data: [{ id: 'mail-owner' }] })
    const or = main?.filters.find(f => f[1] === 'or')?.[2] as string
    expect(or).toContain('and(title.ilike')
    expect(or).toContain('user_id.in.(mail-owner)')
    const uuid = (await open({ q: LISTING })).main?.filters.find(f => f[1] === 'or')?.[2] as string
    expect(uuid).toContain(`id.eq.${LISTING}`)
    expect((await open({ q: 'word' })).main?.filters.find(f => f[1] === 'or')?.[2]).not.toContain('user_id.in')
  })
})
