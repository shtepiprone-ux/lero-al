38	   * `error` is the field's own message (a wrong password). The field takes the focus when the dialog opens.

50	/**
51	 * The one confirmation step of a destructive action (Task 859 Revision 2, R28, owner decision D78-18). It is a
52	 * structured `MantineModal` (bottom sheet below 640) with a section that states the consequence, and a
53	 * `MantineDialogFooter` pair: Cancel on the left (secondary) and the red destructive primary on the right, which
54	 * shows `loading` while the action runs. A dialog's resting destructive action is subtle red; pressing it opens this
55	 * modal, and only the primary here runs the action. With `typedConfirm` the primary also waits for the typed word; with `passwordConfirm` (Task 923 R80) it waits for
56	 * the password instead, and `googleConfirm` adds the way for an account with no password.
57	 */
58	export function MantineDeleteConfirmModal({
59	  opened,
60	  onClose,
61	  title,
62	  message,
63	  confirmLabel,
64	  cancelLabel,
65	  onConfirm,
66	  loading = false,
67	  confirmDisabled = false,
68	  confirmTestId,
69	  bodyTestId,
70	  typedConfirm,
71	  passwordConfirm,
72	  googleConfirm,
73	  children,
74	}: MantineDeleteConfirmModalProps) {
75	  const [typed, setTyped] = useState('')
76	  const [passwordVisible, setPasswordVisible] = useState(false)
77	  // The field starts empty every time the dialog opens: a word typed before a cancel never carries over (Task 923 R46); a password is never left visible either.
78	  useEffect(() => {
79	    if (opened) {
80	      setTyped('')
81	      setPasswordVisible(false)
82	    }
83	  }, [opened])
84	  const typedOk = !typedConfirm || typed.trim().toUpperCase() === typedConfirm.word.toUpperCase()
85	  const passwordOk = !passwordConfirm || passwordConfirm.value !== ''
86	
87	  return (
88	    <MantineModal
89	      structured
90	      opened={opened}
91	      onClose={onClose}
92	      title={title}
93	      footer={
94	        <MantineDialogFooter
95	          secondary={<Button type="button" variant="default" disabled={loading} onClick={onClose}>{cancelLabel}</Button>}
96	          primary={
97	            <Button type="button" color="red" loading={loading} disabled={confirmDisabled || !typedOk || !passwordOk} onClick={onConfirm} data-testid={confirmTestId}>
98	              {confirmLabel}
99	            </Button>
100	          }
101	        />
102	      }
103	    >
104	      <MantineDialogSections>
105	        <MantineDialogSection>
106	          <Box data-testid={bodyTestId}>{typeof message === 'string' ? <Text fz="sm">{message}</Text> : message}</Box>
107	        </MantineDialogSection>
108	        {children}
109	        {googleConfirm && (
110	          <MantineDialogSection>
111	            <Stack gap="sm">
112	              <Text fz="sm">{googleConfirm.message}</Text>
113	              <Button type="button" variant="default" fullWidth disabled={loading} onClick={googleConfirm.onConfirm}>{googleConfirm.label}</Button>
114	            </Stack>
115	          </MantineDialogSection>
116	        )}
117	        {typedConfirm && (
118	          <MantineDialogSection>
119	            <TextInput
120	              label={typedConfirm.label}
121	              placeholder={typedConfirm.word}
122	              autoComplete="off"
123	              data-autofocus
124	              value={typed}
125	              disabled={loading}
