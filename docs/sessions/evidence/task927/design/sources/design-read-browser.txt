     1	/**
     2	 * Sanitizes a `next` redirect path to ensure it is a safe same-origin relative path.
     3	 * (`next` is the canonical post-login redirect param across the whole codebase.)
     4	 *
     5	 * Accepts: relative paths that start with a single "/" (e.g. "/sq/cabinet").
     6	 * Rejects:
     7	 *   - undefined / null / empty string
     8	 *   - protocol-relative URLs (//)
     9	 *   - anything with a URL scheme (javascript:, data:, https:, etc.)
    10	 *   - paths that do not start with /
    11	 *   - backslashes (browser normalisation may expand \ to / in some contexts)
    12	 *   - ASCII control characters (U+0000–U+001F)
    13	 *   - path-traversal segments: ".." in raw or percent-encoded forms
    14	 *     (%2e%2e, .%2e, %2e.) including with encoded slash (%2f)
    15	 *
    16	 * Admin-path enforcement (after login): the server-side admin layout guard
    17	 * handles unauthorized access — a non-admin who lands on /admin is
    18	 * immediately redirected away by the role check in app/admin/layout.tsx.
    19	 * sanitizeReturnTo is a path-safety check only, NOT an authorization check.
    20	 */
    21	export function sanitizeReturnTo(raw: string | null | undefined): string | null {
    22	  if (!raw) return null
    23	  // Must start with / but not //
    24	  if (!raw.startsWith('/') || raw.startsWith('//')) return null
    25	  // Must not contain a URL scheme (javascript:, data:, vbscript:, etc.)
    26	  if (/^[a-z][a-z0-9+.\-]*:/i.test(raw)) return null
    27	  // Reject backslashes — browsers may normalise \ to / in some contexts.
    28	  if (raw.includes('\\')) return null
    29	  // Reject ASCII control characters (U+0000–U+001F).
    30	  if (/[\x00-\x1f]/.test(raw)) return null
    31	  // Reject path-traversal segments. Normalise %2e → . so that all encoded
    32	  // forms (%2e%2e, .%2e, %2e.) collapse to "..", then test for ".." as a
    33	  // path segment (followed by /, end-of-string, ?, #, or encoded slash %2f).
    34	  const normalized = raw.replace(/%2e/gi, '.')
    35	  if (/(^|\/)\.\.($|[/?#]|%2f)/i.test(normalized)) return null
    36	  return raw
    37	}
